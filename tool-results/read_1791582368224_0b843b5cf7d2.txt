    64→function sampleColor(
    65→  ctx: CanvasRenderingContext2D,
    66→  x: number,
    67→  y: number,
    68→  w: number,
    69→  h: number,
    70→  pw: number,
    71→  ph: number
    72→): string {
    73→  const sx = Math.max(0, Math.floor(x * 2))
    74→  const sy = Math.max(0, Math.floor(y * 2))
    75→  const sw = Math.min(Math.ceil(w * 2), Math.floor(pw * 2) - sx)
    76→  const sh = Math.min(Math.ceil(h * 2), Math.floor(ph * 2) - sy)
    77→  if (sw <= 2 || sh <= 2) return '#111111'
    78→  let img: ImageData
    79→  try {
    80→    img = ctx.getImageData(sx, sy, sw, sh)
    81→  } catch {
    82→    return '#111111'
    83→  }
    84→  const counts = new Map<number, number>()
    85→  const d = img.data
    86→  for (let i = 0; i < d.length; i += 4) {
    87→    const r = d[i], g = d[i + 1], b = d[i + 2], a = d[i + 3]
    88→    if (a < 50) continue
    89→    if (r > 205 && g > 205 && b > 205) continue
    90→    const key = ((r >> 4) << 8) | ((g >> 4) << 4) | (b >> 4)
    91→    counts.set(key, (counts.get(key) || 0) + 1)
    92→  }
    93→  let bc = 0
    94→  for (const [, c] of counts) bc = Math.max(bc, c)
    95→  // من بين الألوان المتكررة بشكل معقول نختار الأغمق (لب الضرفة) لا الأكثر تكرارًا (حواف التنعيم)
    96→  let best = -1
    97→  let bestLum = Infinity
    98→  for (const [k, c] of counts) {
    99→    if (c < Math.max(4, bc * 0.12)) continue
   100→    const r = ((k >> 8) & 15) * 16 + 8
   101→    const g = ((k >> 4) & 15) * 16 + 8
   102→    const b = (k & 15) * 16 + 8
   103→    const lum = 0.2126 * r + 0.7152 * g + 0.0722 * b
   104→    if (lum < bestLum) {
   105→      bestLum = lum
   106→      best = k
   107→    }
   108→  }
   109→  if (best < 0 || bc < 8) return '#111111'
   110→  const r = ((best >> 8) & 15) * 16 + 8
   111→  const g = ((best >> 4) & 15) * 16 + 8
   112→  const b = (best & 15) * 16 + 8
   113→  return '#' + [r, g, b].map((v) => v.toString(16).padStart(2, '0')).join('')
   114→}

   116→async function loadPdfIntoPages(
   117→  file: File,
   118→  onProgress: (s: string) => void
   119→): Promise<DocData> {
   120→  const pdfjs = await import('pdfjs-dist')
   121→  pdfjs.GlobalWorkerOptions.workerSrc = '/pdf.worker.min.mjs'
   122→  const buf = await file.arrayBuffer()
   123→  const doc = await pdfjs.getDocument({
   124→    data: new Uint8Array(buf.slice(0)),
   125→    standardFontDataUrl: '/pdfjs-std/',
   126→    cMapUrl: '/pdfjs-cmaps/',
   127→    cMapPacked: true,
   128→    isEvalSupported: false,
   129→  }).promise
   130→
   131→  const pages: PageData[] = []
   132→  for (let pn = 1; pn <= doc.numPages; pn++) {
   133→    onProgress(`جاري تجهيز الصفحة ${pn} من ${doc.numPages}…`)
   134→    const page = await doc.getPage(pn)
   135→    const p1 = page.getViewport({ scale: 1 })
   136→    const S = 794 / p1.width
   137→    const vp = page.getViewport({ scale: S })
   138→    const c2 = page.getViewport({ scale: S * 2 })
   139→
   140→    const canvas = document.createElement('canvas')
   141→    canvas.width = Math.floor(c2.width)
   142→    canvas.height = Math.floor(c2.height)
   143→    const ctx = canvas.getContext('2d', { willReadFrequently: true })!
   144→    await page.render({ canvasContext: ctx, viewport: c2 }).promise
   145→    const bg = canvas.toDataURL('image/png')
   146→
   147→    const tc = await page.getTextContent()
   148→    const items: GeoItem[] = []
   149→    for (const it of tc.items as unknown as Record<string, unknown>[]) {
   150→      const str = it.str as string
   151→      if (!str || !str.trim()) continue
   152→      const tr = it.transform as number[]
   153→      const tx = pdfjs.Util.transform(vp.transform, tr)
   154→      const fh = Math.hypot(tx[2], tx[3])
   155→      if (fh < 3) continue
   156→      items.push({
   157→        x: tx[4],
   158→        y: tx[5],
   159→        w: ((it.width as number) || 0) * S,
   160→        h: fh,
   161→        str,
   162→      })
   163→    }
   164→
   165→    const width = Math.round(p1.width * S)
   166→    const height = Math.round(p1.height * S)
   167→    const lines = buildLineBlocks(items)
   168→    const blocks: TextBlock[] = lines.map((l, i) => {
   169→      const patchBox = {
   170→        x: Math.max(0, l.x - 3),
   171→        y: Math.max(0, l.baseline - l.size * 1.08),
   172→        w: l.right - l.x + 6,
   173→        h: l.size * 1.55,
   174→      }
   175→      if (patchBox.y + patchBox.h > height) patchBox.h = height - patchBox.y
   176→      return {
   177→        id: `p${pn}-${i}`,
   178→        page: pn,
   179→        x: l.x,
   180→        top: l.baseline - l.size * BASELINE_K,
   181→        w: l.right - l.x,
   182→        size: l.size,
   183→        baseline: l.baseline,
   184→        text: l.text,
   185→        color: sampleColor(ctx, l.x, l.baseline - l.size * 1.05, l.right - l.x, l.size * 1.5, width, height),
   186→        origin: 'pdf' as const,
   187→        patchBox,
   188→        dirty: false,
   189→        deleted: false,
   190→        rtl: AR_RE.test(l.text),
   191→      }
   192→    })
   193→    pages.push({ num: pn, width, height, bg, blocks, initial: cloneBlocks(blocks) })
   194→  }
   195→
   196→  return { fileName: file.name || 'document.pdf', pages }
   197→}

   201→interface BlockViewProps {
   202→  b: TextBlock
   203→  z: number
   204→  selected: boolean
   205→  editing: boolean
   206→  addMode: boolean
   207→  onSelect: (id: string) => void
   208→  onBeginEdit: (id: string) => void
   209→  onEndEdit: (id: string, text: string) => void
   210→  onMove: (id: string, x: number, top: number) => void
   211→  onEndMove: () => void
   212→}

   214→function BlockView({
   215→  b,
   216→  z,
   217→  selected,
   218→  editing,
   219→  addMode,
   220→  onSelect,
   221→  onBeginEdit,
   222→  onEndEdit,
   223→  onMove,
   224→  onEndMove,
   225→}: BlockViewProps) {
   226→  const elRef = useRef<HTMLDivElement>(null)
   227→  const drag = useRef<{ cx: number; cy: number; ox: number; oy: number; moved: boolean } | null>(null)
   228→  const liveText = useRef(b.text)
   229→
   230→  useEffect(() => {
   231→    if (editing && elRef.current) {
   232→      const el = elRef.current
   233→      el.textContent = b.text
   234→      liveText.current = b.text
   235→      el.focus()
   236→      const range = document.createRange()
   237→      range.selectNodeContents(el)
   238→      range.collapse(false)
   239→      const sel = window.getSelection()
   240→      sel?.removeAllRanges()
   241→      sel?.addRange(range)
   242→    }
   243→  }, [editing])
   244→
   245→  const onPointerDown = (e: React.PointerEvent) => {
   246→    if (editing || addMode) return
   247→    e.stopPropagation()
   248→    ;(e.currentTarget as HTMLElement).setPointerCapture(e.pointerId)
   249→    drag.current = { cx: e.clientX, cy: e.clientY, ox: b.x, oy: b.top, moved: false }
   250→    onSelect(b.id)
   251→  }
   252→
   253→  const onPointerMove = (e: React.PointerEvent) => {
   254→    const d = drag.current
   255→    if (!d || editing) return
   256→    const dx = (e.clientX - d.cx) / z
   257→    const dy = (e.clientY - d.cy) / z
   258→    if (!d.moved && Math.abs(dx) + Math.abs(dy) > 3) d.moved = true
   259→    if (d.moved) onMove(b.id, d.ox + dx, d.oy + dy)
   260→  }
   261→
   262→  const onPointerUp = () => {
   263→    const d = drag.current
   264→    drag.current = null
   265→    if (d?.moved) onEndMove()
   266→  }
   267→
   268→  return (
   269→    <div
   270→      ref={elRef}
   271→      data-bid={b.id}
   272→      className={cn('blk', selected && !editing && 'sel', editing && 'edit')}
   273→      dir={b.rtl ? 'rtl' : 'ltr'}
   274→      style={{
   275→        left: b.x,
   276→        top: b.top,
   277→        width: b.w,
   278→        fontSize: b.size,
   279→        color: editing || b.dirty || b.origin === 'new' ? b.color : 'transparent',
   280→        fontFamily: AMIRI_STACK,
   281→        textAlign: b.rtl ? 'right' : 'left',
   282→      }}
   283→      contentEditable={editing}
   284→      suppressContentEditableWarning
   285→      spellCheck={false}
   286→      onPointerDown={onPointerDown}
   287→      onPointerMove={onPointerMove}
   288→      onPointerUp={onPointerUp}
   289→      onDoubleClick={(e) => {
   290→        e.stopPropagation()
   291→        if (!editing) onBeginEdit(b.id)
   292→      }}
   293→      onInput={(e) => {
   294→        liveText.current = (e.currentTarget as HTMLElement).innerText
   295→      }}
   296→      onBlur={() => {
   297→        if (editing) onEndEdit(b.id, liveText.current)
   298→      }}
   299→      onKeyDown={(e) => {
   300→        if (e.key === 'Escape') (e.currentTarget as HTMLElement).blur()
   301→        e.stopPropagation()
   302→      }}
   303→    >
   304→      {!editing ? b.text : null}
   305→    </div>
   306→  )
   307→}

   562→      })
